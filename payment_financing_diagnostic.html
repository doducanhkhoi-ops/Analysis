QUERY 1: TỔNG QUAN 4 PHƯƠNG THỨC THANH TOÁN (BẢNG 1)
SELECT 
    p.payment_type,
    COUNT(DISTINCT o.order_id) AS total_orders,
    ROUND(SUM(CAST(p.payment_value AS DECIMAL(10,2))), 2) AS total_payment_value,
    ROUND(AVG(CAST(p.payment_value AS DECIMAL(10,2))), 2) AS avg_payment_value,
    ROUND(AVG(CAST(p.payment_installments AS SIGNED)), 1) AS avg_installments,
    ROUND(AVG(CASE WHEN o.order_status = 'delivered' AND o.order_delivered_customer_date IS NOT NULL 
        THEN DATEDIFF(STR_TO_DATE(o.order_approved_at, '%Y-%m-%d'), STR_TO_DATE(o.order_purchase_timestamp, '%Y-%m-%d')) END), 2) AS avg_approval_days,
    ROUND(AVG(CASE WHEN o.order_status = 'delivered' AND o.order_delivered_customer_date IS NOT NULL 
        THEN DATEDIFF(STR_TO_DATE(o.order_delivered_customer_date, '%Y-%m-%d'), STR_TO_DATE(o.order_purchase_timestamp, '%Y-%m-%d')) END), 1) AS avg_delivery_days,
    ROUND(AVG(CAST(r.review_score AS DECIMAL(3,2))), 2) AS avg_review_score,
    ROUND(100.0 * SUM(CASE WHEN CAST(r.review_score AS SIGNED) = 1 THEN 1 ELSE 0 END) / COUNT(r.review_id), 2) AS pct_1_star,
    ROUND(100.0 * SUM(CASE WHEN o.order_status = 'canceled' THEN 1 ELSE 0 END) / COUNT(DISTINCT o.order_id), 2) AS cancel_rate_pct
FROM olist_raw.payments p
JOIN olist_raw.orders o ON p.order_id = o.order_id
LEFT JOIN olist_raw.reviews r ON o.order_id = r.order_id
WHERE p.payment_type != 'not_defined'
GROUP BY p.payment_type
ORDER BY total_payment_value DESC;
QUERY 2: MA TRẬN 2 CHIỀU SLA x KỲ HẠN (BẢNG 4)
SELECT 
    CASE 
        WHEN CAST(p.payment_installments AS SIGNED) = 1 THEN '1 kỳ'
        WHEN CAST(p.payment_installments AS SIGNED) BETWEEN 2 AND 5 THEN '2 - 5 kỳ'
        WHEN CAST(p.payment_installments AS SIGNED) BETWEEN 6 AND 9 THEN '6 - 9 kỳ'
        WHEN CAST(p.payment_installments AS SIGNED) = 10 THEN '10 kỳ'
        ELSE '>= 11 kỳ'
    END AS installment_group,
    CASE 
        WHEN STR_TO_DATE(o.order_delivered_customer_date, '%Y-%m-%d') > STR_TO_DATE(o.order_estimated_delivery_date, '%Y-%m-%d') THEN 'Late (Trễ hạn)'
        ELSE 'On-time (Đúng hạn)'
    END AS delivery_sla_status,
    COUNT(DISTINCT o.order_id) AS order_count,
    ROUND(AVG(CAST(r.review_score AS DECIMAL(3,2))), 2) AS avg_review,
    ROUND(100.0 * SUM(CASE WHEN CAST(r.review_score AS SIGNED) = 1 THEN 1 ELSE 0 END) / COUNT(r.review_id), 2) AS pct_1_star
FROM olist_raw.payments p
JOIN olist_raw.orders o ON p.order_id = o.order_id
LEFT JOIN olist_raw.reviews r ON o.order_id = r.order_id
WHERE p.payment_type = 'credit_card'
  AND o.order_status = 'delivered'
  AND o.order_delivered_customer_date IS NOT NULL
GROUP BY installment_group, delivery_sla_status
ORDER BY installment_group, delivery_sla_status;
