QUERY_01_MONTHLY_LOGISTICS_TREND.SQL 
SELECT 
    DATE_FORMAT(STR_TO_DATE(o.order_purchase_timestamp, '%Y-%m-%d %H:%i:%s'), '%Y-%m') AS order_month,
    COUNT(DISTINCT o.order_id) AS total_orders,
    SUM(CASE WHEN o.order_status = 'delivered' THEN 1 ELSE 0 END) AS delivered_orders,
    ROUND(SUM(CASE WHEN o.order_status = 'delivered' THEN p.payment_value ELSE 0 END), 2) AS delivered_gmv_brl,
    ROUND(AVG(CASE WHEN o.order_status = 'delivered' AND o.order_delivered_customer_date IS NOT NULL 
        THEN DATEDIFF(STR_TO_DATE(o.order_delivered_customer_date, '%Y-%m-%d'), STR_TO_DATE(o.order_purchase_timestamp, '%Y-%m-%d')) END), 1) AS avg_delivery_days,
    ROUND(AVG(CASE WHEN o.order_status = 'delivered' AND o.order_estimated_delivery_date IS NOT NULL 
        THEN DATEDIFF(STR_TO_DATE(o.order_estimated_delivery_date, '%Y-%m-%d'), STR_TO_DATE(o.order_purchase_timestamp, '%Y-%m-%d')) END), 1) AS avg_estimated_days,
    ROUND(100.0 * SUM(CASE WHEN o.order_status = 'delivered' AND STR_TO_DATE(o.order_delivered_customer_date, '%Y-%m-%d') > STR_TO_DATE(o.order_estimated_delivery_date, '%Y-%m-%d') THEN 1 ELSE 0 END) 
        / NULLIF(SUM(CASE WHEN o.order_status = 'delivered' THEN 1 ELSE 0 END), 0), 2) AS late_delivery_pct
FROM olist_raw.orders o
LEFT JOIN olist_raw.payments p ON o.order_id = p.order_id
WHERE o.order_purchase_timestamp >= '2017-09-01' AND o.order_purchase_timestamp <= '2018-02-28'
GROUP BY order_month
ORDER BY order_month;
QUERY_02_PIPELINE_STAGES_DECOMPOSITION.SQL 
SELECT 
    CASE 
        WHEN STR_TO_DATE(order_purchase_timestamp, '%Y-%m-%d') BETWEEN '2017-10-15' AND '2017-11-12' THEN '1. Normal Period'
        WHEN STR_TO_DATE(order_purchase_timestamp, '%Y-%m-%d') BETWEEN '2017-11-20' AND '2017-12-03' THEN '2. Black Friday Peak'
        WHEN STR_TO_DATE(order_purchase_timestamp, '%Y-%m-%d') BETWEEN '2017-12-04' AND '2017-12-31' THEN '3. Holiday Season'
    END AS period_name,
    COUNT(*) AS total_delivered_orders,
    ROUND(AVG(DATEDIFF(STR_TO_DATE(order_approved_at, '%Y-%m-%d'), STR_TO_DATE(order_purchase_timestamp, '%Y-%m-%d'))), 2) AS avg_days_to_approve,
    ROUND(AVG(DATEDIFF(STR_TO_DATE(order_delivered_carrier_date, '%Y-%m-%d'), STR_TO_DATE(order_approved_at, '%Y-%m-%d'))), 2) AS avg_days_to_carrier,
    ROUND(AVG(DATEDIFF(STR_TO_DATE(order_delivered_customer_date, '%Y-%m-%d'), STR_TO_DATE(order_delivered_carrier_date, '%Y-%m-%d'))), 2) AS avg_days_carrier_to_customer,
    ROUND(AVG(DATEDIFF(STR_TO_DATE(order_delivered_customer_date, '%Y-%m-%d'), STR_TO_DATE(order_purchase_timestamp, '%Y-%m-%d'))), 2) AS avg_total_delivery_days,
    ROUND(100.0 * SUM(CASE WHEN STR_TO_DATE(order_delivered_customer_date, '%Y-%m-%d') > STR_TO_DATE(order_estimated_delivery_date, '%Y-%m-%d') THEN 1 ELSE 0 END) / COUNT(*), 2) AS late_delivery_pct
FROM olist_raw.orders
WHERE order_status = 'delivered'
  AND order_delivered_customer_date IS NOT NULL 
  AND order_delivered_carrier_date IS NOT NULL
  AND order_approved_at IS NOT NULL
  AND order_purchase_timestamp >= '2017-10-15' AND order_purchase_timestamp <= '2017-12-31'
GROUP BY period_name
HAVING period_name IS NOT NULL;
